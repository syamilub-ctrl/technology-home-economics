📌 구글 시트 Apps Script 코드 (Code.gs):

구글 시트 [확장 프로그램] > [Apps Script]에 아래 코드가 저장 및 [웹 앱 / 모든 사용자]로 배포되어 있으면 모든 학생 기기가 즉시 연동됩니다.

function doGet(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var params = (e && e.parameter) ? e.parameter : {};
  var action = params.action || "read";
  var callback = params.callback || "";

  // 1. 데이터 기록 (CORS 우회를 위해 GET 파라미터 수신)
  if (action === "write") {
    if (sheet.getLastRow() === 0) {
      sheet.appendRow(["시간", "반", "학번이름(교사용)", "검사종류", "결과유형", "세부점수"]);
    }
    sheet.appendRow([
      new Date().toLocaleString("ko-KR"),
      params.studentClass || "미지정",
      params.studentIdName || "익명",
      params.testType || "",
      params.resultType || "",
      params.detailScores || ""
    ]);
    return makeResponse({ status: "ok" }, callback);
  }

  // 2. 실시간 익명 통계 조회
  var rows = sheet.getDataRange().getValues();
  var list = [];
  for (var i = 1; i < rows.length; i++) {
    // 개인정보 보호를 위해 학번이름(rows[i][2])은 절대 외부로 반환하지 않음
    list.push({
      studentClass: String(rows[i][1] || ""),
      testType: String(rows[i][3] || ""),
      resultType: String(rows[i][4] || ""),
      detailScores: String(rows[i][5] || "")
    });
  }
  return makeResponse({ status: "ok", records: list }, callback);
}

function makeResponse(obj, callback) {
  var jsonStr = JSON.stringify(obj);
  if (callback) {
    return ContentService
      .createTextOutput(callback + "(" + jsonStr + ")")
      .setMimeType(ContentService.MimeType.JAVASCRIPT);
  }
  return ContentService
    .createTextOutput(jsonStr)
    .setMimeType(ContentService.MimeType.JSON);
}
