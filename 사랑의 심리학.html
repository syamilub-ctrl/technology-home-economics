✍️ 작성 길잡이 (이론 기반 예시):
“나의 검사 결과 스턴버그의 삼각형에서는 ‘친밀감(7.8점)’이 가장 높게 나타났고, 리의 색깔 이론에서는 우정적 사랑인 ‘스토르게’와 헌신적 사랑인 ‘아가페’가 1·2순위로 나타났다. 이는 내가 일시적인 설렘(열정)보다 서로를 깊이 이해하고 신뢰하는 정서적 유대를 소중히 여긴다는 강점을 보여준다. 반면 ‘헌신’ 점수가 상대적으로 낮게 나온 것은, 갈등이 생겼을 때 끝까지 책임감을 가지고 관계를 지켜내려는 의지를 더 길러야 함을 시사한다.”

📌 3단계 초간단 연동 방법:

1. 새 구글 스프레드시트를 열고 상단 메뉴 [확장 프로그램] > [Apps Script]를 클릭합니다.

2. 아래 코드를 복사해 붙여넣고 저장(Ctrl+S)한 뒤, 우측 상단 [배포] > [새 배포]를 누릅니다.

3. 유형을 [웹 앱]으로 선택하고 액세스 권한을 '모든 사용자(Anyone)'로 배포한 뒤 나온 URL을 위 칸에 저장하세요.

function doPost(e) {
  if (!e || !e.postData) {
    return ContentService.createTextOutput(JSON.stringify({ status: "no_data" }))
      .setMimeType(ContentService.MimeType.JSON);
  }
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  if (sheet.getLastRow() === 0) {
    sheet.appendRow(["시간", "반", "학번이름(교사용)", "검사종류", "결과유형", "세부점수", "점수데이터(JSON)"]);
  }
  var data = JSON.parse(e.postData.contents);
  sheet.appendRow([
    new Date().toLocaleString("ko-KR"),
    data.studentClass || "미지정",
    data.studentIdName || "익명",
    data.testType,
    data.resultType,
    data.detailScores,
    data.scoresJson || ""
  ]);
  return ContentService.createTextOutput(JSON.stringify({ status: "ok" }))
    .setMimeType(ContentService.MimeType.JSON);
}

function doGet(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var rows = sheet.getDataRange().getValues();
  var list = [];
  for (var i = 1; i < rows.length; i++) {
    var parsedScores = null;
    if (rows[i][6]) {
      try { parsedScores = JSON.parse(rows[i][6]); } catch (err) {}
    }
    // 개인정보 보호를 위해 학번이름(rows[i][2])은 절대 반환하지 않음
    list.push({
      studentClass: rows[i][1],
      testType: rows[i][3],
      resultType: rows[i][4],
      scores: parsedScores
    });
  }
  return ContentService.createTextOutput(JSON.stringify(list))
    .setMimeType(ContentService.MimeType.JSON);
}
